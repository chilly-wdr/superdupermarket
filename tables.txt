staff:
CREATE TABLE STAFF(staff_id INT PRIMARY KEY,staff_name VARCHAR(20),mobile_number INT);

rewards:
CREATE TABLE REWARDS(customer_id INT PRIMARY KEY,customer_name VARCHAR(30) NOT NULL,mobile_number INT UNIQUE,DOJ DATE,reward_points INT);

stock:
CREATE TABLE STOCK(product_id INT PRIMARY KEY,product_name VARCHAR(30) UNIQUE,stock_available INT NOT NULL,last_restock_date DATE NOT NULL,market_price FLOAT NOT NULL,price FLOAT NOT NULL);

discount:
CREATE TABLE DISCOUNT(discount_id INT PRIMARY KEY,product_id INT,discount_percentage INT,discounted_price FLOAT,start_date DATE NOT NULL,end_date DATE NOT NULL,FOREIGN KEY (product_id) REFERENCES stock(product_id));

billing:
CREATE TABLE BILLING(bill_id INT PRIMARY KEY,date_of_transaction DATE NOT NULL,time_of_transaction TIME NOT NULL,staff_id INT NOT NULL,no_of_items INT NOT NULL,total_amount FLOAT NOT NULL,reward_customer CHAR(1),customer_mobile_number int(8),issued_reward_points INT,reward_points_redeemed INT,FOREIGN KEY (staff_id) REFERENCES staff(staff_id));

transaction:
CREATE TABLE TRANSACTION(bill_id INT,product_id INT,qty INT,unit_price FLOAT,total_price FLOAT,FOREIGN KEY (product_id) REFERENCES stock(product_id));
